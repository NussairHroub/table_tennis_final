% Purpose: Customizes header, footer, and title page

% ---- Header and Footer Setup ----
\pagestyle{fancy}
\fancyhf{}

\fancyhead[L]{\textsc{\headerText}}
\fancyhead[R]{\textit{\nouppercase{\studentName}}}
\fancyfoot[C]{\thepage}

\renewcommand{\headrulewidth}{0.4pt}
\renewcommand{\footrulewidth}{0pt}

\fancypagestyle{plain}{
  \fancyhf{}
  \fancyfoot[C]{\thepage}
  \renewcommand{\headrulewidth}{0pt}
  \renewcommand{\footrulewidth}{0pt}
}
